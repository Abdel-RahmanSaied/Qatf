# Word substitutions applied to caption text after transcription.
# Format: wrong = right   (or  wrong -> right).  '#' comments, blanks ignored.
#
# Only for errors the vocabulary demonstrably will not fix. Prefer --vocab:
# it addresses the cause and generalises. Every entry here is a word Whisper
# kept getting wrong even with the correct spelling in the vocabulary list.
#
# Timestamps are never modified — this changes what a caption reads, never
# where a clip is cut.

# Python: the speaker says it closer to "bayson", so the model is not
# mishearing — it is spelling what it heard. Vocabulary could not override it.
بايسون = بايثون
بايسن = بايثون

# ث -> س, how the dialect pronounces it. The vocabulary catches most, not all.
مسلا = مثلا
بسلا = مثلا

# AI read as two Arabic letters rather than the term
الاي = الذكاء

# platform
المنصبة = المنصة

# --- from the "متسلمش دماغك" pass (12-min 4K, car audio) -------------------
#
# Every rule below was verified to match a real token in that transcript before
# being added. A rule that matches nothing is worse than no rule: it reads as
# coverage and provides none. Four of the five rules above match zero tokens in
# that file — they are kept because they earned their place on other material,
# not because they fire here.

# THE ONE THAT MATTERS. Whisper produced one token where the speaker said two,
# and the collapse INVERTS the meaning: the clip's whole point is that the
# person who was hurt does NOT forget. Substituting two words into one token is
# the sanctioned move — `edits.diff` names it in its own error message — and it
# leaves the word count and every timing untouched, so no cut can move.
هممكن = مش ممكن

# ق -> ء / ك, and hamza dropped. Ordinary dialect, none of these are real words,
# so a whole-token rule cannot damage a correct occurrence elsewhere.
العصد = القصد
قويسة = كويسة
تنشأمش = تتشاءمش
أرأة = آراء
الأجال = الأجيال

# heard as place names on noisy car audio — أمانة (trust) became عمانة
عمانة = أمانة

# only ever appears as "سراحة." with the stop; apply() strips trailing
# punctuation before matching and puts it back, so this catches it
سراحة = بصراحة

# --- X (Twitter), measured on the 30-term-vocabulary transcript ------------
# The vocabulary does not take these: with `إكس` in the hotword list the model
# starts DROPPING other speech (measured: uncovered speech 66s -> 167s), so the
# term is corrected here instead, where it costs nothing.
اكسة = إكس
اكس = إكس
